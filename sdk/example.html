Use it

    <script type="module">
  import { NimNames } from 'https://signal.heylogram.tv/kovanapi/nim.js';
  const nim = NimNames({ app: 'mygame' });   // your own .nim name
  const who = await nim.login();             // { address, name }
  await nim.pay('mando', 1);                 // 1 NIM to mando.nim
</script>
